\else
  \newcommand{\FontFamilySelection}{modern}
  \ClassWarning{MyDocument}{%
    No \protect\FontFamilySelection\space defined externally.\MessageBreak
    Falling back to default: 'modern' for font Latin Modern%
  }
\fi

\makeatletter
\ifnum\pdf@strcmp{\FontFamilySelection}{libertinus}=0
  % Set the fonts to Libertinus
  \setmainfont{Libertinus Serif}
  \setsansfont{Libertinus Sans}
  \setmonofont{Libertinus Mono}
  \setmathfont{Libertinus Math}
\else
  \ifnum\pdf@strcmp{\FontFamilySelection}{noto}=0
    % Set fonts to Noto
    \setmainfont{Noto Serif}
    \setsansfont{Noto Sans}
    \setmonofont{Noto Sans Mono}
    \setmathfont{Noto Sans Math}
  \else
    \ifnum\pdf@strcmp{\FontFamilySelection}{dejavu}=0
      % Set fonts to DejaVu
      \setmainfont{DejaVu Serif}
      \setsansfont{DejaVu Sans}
      \setmonofont{DejaVu Sans Mono}      %[Scale=MatchLowercase]
      \setmathfont{DejaVu Math TeX Gyre}  %DejaVuMathTeXGyre-Regular
    \else
      % Set the fonts to Latin Modern (LaTeX font derived from Computer Modern)
      \setmainfont{Latin Modern Roman}
      \setsansfont{Latin Modern Sans}
      \setmonofont{Latin Modern Mono}
      \setmathfont{Latin Modern Math}
    \fi
  \fi
\fi
\makeatother
